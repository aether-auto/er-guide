```typescript
import type { MapCode, MapRef } from './types'

// VERIFIED 2026-06-09: Fetched live from eldenring.wiki.fextralife.com.
// Overworld deep links: Interactive+Map?id=<id>&lat=<lat>&lng=<lng>&code=mapA
// Underground: code=mapB; Ashen: code=mapC; DLC: code=mapD
// zoom param is NOT used in actual wiki item deep links.

const BASE_MAP = 'https://eldenring.wiki.fextralife.com/Interactive+Map'

const MAP_PAGE: Record<MapCode, string> = {
  overworld: BASE_MAP,
  underground: BASE_MAP,
  ashen: BASE_MAP,
  dlc: BASE_MAP,
}

const MAP_CODE: Record<MapCode, string> = {
  overworld: 'mapA',
  underground: 'mapB',
  ashen: 'mapC',
  dlc: 'mapD',
}

export function mapUrl(ref: MapRef): string {
  const url = new URL(MAP_PAGE[ref.code])
  if (ref.markerId != null) url.searchParams.set('id', String(ref.markerId))
  url.searchParams.set('lat', String(ref.lat))
  url.searchParams.set('lng', String(ref.lng))
  url.searchParams.set('code', MAP_CODE[ref.code])
  return url.toString()
}

// The external Fextralife map link — opens in a named companion window from popups.
export const FEXTRALIFE_MAP_URL = MAP_PAGE.overworld
```

```typescript
import type { Item, Leg, MapCode } from './types'

/**
 * Returns ordered [lat, lng] pairs for item steps in `leg` whose item has
 * a map reference on the given `code` layer. Steps without a map ref or on
 * a different layer are silently skipped. Non-item steps (direction, boss,
 * quest) are always skipped.
 *
 * Pure function — no side effects, safe to call in tests and workers.
 */
export function routePathForLayer(
  leg: Leg,
  itemsById: Map<string, Item>,
  code: MapCode,
): [number, number][] {
  const coords: [number, number][] = []
  for (const step of leg.steps) {
    if (step.type !== 'item') continue
    const item = itemsById.get(step.itemId)
    if (!item?.map) continue
    if (item.map.code !== code) continue
    coords.push([item.map.lat, item.map.lng])
  }
  return coords
}
```

Do NOT re-project — trust the coordinates as-is; they match what the Fextralife map app uses.

```typescript
import L from 'leaflet'
import 'leaflet/dist/leaflet.css'
import type { MapCode, MapRef } from '../lib/types'
import { categoryIcon, graceIcon, type MarkerVariant } from '../lib/markers'
import { routePathForLayer } from '../lib/route-path'
import { mapUrl, FEXTRALIFE_MAP_URL } from '../lib/maplink'
import { CATEGORY_META } from '../lib/types'
import { itemsById, regions } from '../lib/data'
import { progressStore } from '../lib/useProgress'
import { useUi } from '../App'
import mapExtras from '../../data/map-extras.json'

// ── Types ──────────────────────────────────────────────────────────────────

interface GraceEntry { code: string; markerId: number; name: string; lat: number; lng: number }
const graces = (mapExtras as { graces: GraceEntry[] }).graces

// ── Constants ──────────────────────────────────────────────────────────────

const LAYERS: { code: MapCode; label: string }[] = [
  { code: 'overworld',   label: 'Overworld'   },
  { code: 'underground', label: 'Underground' },
  { code: 'ashen',       label: 'Ashen'       },
  { code: 'dlc',         label: 'DLC'         },

// at z0. The marker dataset uses lat ≈ -196..-50, lng ≈ 80..220, which sits inside
// the z0 256x256 tile plane — use these directly.
const MAP_BOUNDS: L.LatLngBoundsLiteral = [[-256, 0], [0, 256]]
const INITIAL_CENTER: L.LatLngExpression = [-128, 128]
const INITIAL_ZOOM = 3

const BASE_URL = import.meta.env.BASE_URL as string

// ── MapView component ──────────────────────────────────────────────────────

export default function MapView({ initialLayer }: { initialLayer?: MapCode }) {
  const { registerFocus } = useUi()
  const containerRef  = useRef<HTMLDivElement>(null)
  // Stable refs so event handlers close over them without stale values
  const mapRef        = useRef<L.Map | null>(null)
  const layerRef      = useRef<MapCode>(initialLayer ?? 'overworld')
  const tileLayerRef  = useRef<L.TileLayer | null>(null)
  const gracesGroupRef= useRef<L.LayerGroup | null>(null)
  const itemGroupRef  = useRef<L.LayerGroup | null>(null)
  const polyGroupRef  = useRef<L.LayerGroup | null>(null)
  const markerMapRef  = useRef<Map<string, L.Marker>>(new Map()) // itemId → marker
  const showGracesRef = useRef(true)
  const tabContainerRef = useRef<HTMLDivElement | null>(null)

  useEffect(() => {
    if (!containerRef.current || mapRef.current) return

    // ── Init map ──────────────────────────────────────────────────────────
    const map = L.map(containerRef.current, {
      crs: L.CRS.Simple,
      center: INITIAL_CENTER,
      zoom: INITIAL_ZOOM,
      minZoom: 0,
      maxZoom: 7,
      maxBounds: MAP_BOUNDS,
      maxBoundsViscosity: 0.8,
      zoomControl: true,
      attributionControl: true,
    })
    mapRef.current = map

    // ── Tile layer ────────────────────────────────────────────────────────
    function tileUrl(code: MapCode) {
      return `${BASE_URL}tiles/${code}/{z}/{x}/{y}.webp`
    }

    const tileLayer = L.tileLayer(tileUrl(layerRef.current), {
      minZoom: 0,
      maxZoom: 7,
      maxNativeZoom: 6,
      noWrap: true,
      tileSize: 256,

    gracesGroupRef.current = gracesGroup
    renderGraces(gracesGroup, layerRef.current)

    // ── Item pins layer ───────────────────────────────────────────────────

    itemGroupRef.current = itemGroup
    renderItems(itemGroup, markerMapRef.current, layerRef.current, map)

    renderPolylines(polyGroup, layerRef.current)

    // ── Layer tabs control ────────────────────────────────────────────────

      onAdd() {
        const div = L.DomUtil.create('div', 'er-layer-controls leaflet-bar')
        div.style.cssText = 'display:flex;flex-direction:column;gap:4px;padding:4px;background:rgba(14,12,9,0.85);border:1px solid #2c261b;border-radius:6px;'
        tabContainerRef.current = div

        LAYERS.forEach(({ code, label }) => {
          const btn = L.DomUtil.create('button', 'er-layer-tab', div) as HTMLButtonElement
          btn.textContent = label
          btn.dataset.code = code
          if (code === layerRef.current) btn.classList.add('active')
          L.DomEvent.on(btn, 'click', (e) => {
            L.DomEvent.stopPropagation(e)
            switchLayer(code)
          })
        })

        // Graces toggle
        const sep = L.DomUtil.create('div', '', div)
        sep.style.cssText = 'border-top:1px solid #2c261b;margin:2px 0;'
        const graceBtn = L.DomUtil.create('button', 'er-layer-tab', div) as HTMLButtonElement
        graceBtn.id = 'er-grace-toggle'
        graceBtn.textContent = '✦ Graces'

          L.DomEvent.stopPropagation(e)
          showGracesRef.current = !showGracesRef.current
          if (showGracesRef.current) {
            gracesGroup.addTo(map)
            graceBtn.classList.add('active')
          } else {
            gracesGroup.remove()
            graceBtn.classList.remove('active')
          }
        })
        return div
      },
    })
    new LayerControl({ position: 'topright' }).addTo(map)

      if (ref.code !== layerRef.current) switchLayer(ref.code)
      map.flyTo([ref.lat, ref.lng], 6, { duration: 0.6 })

      }

        const isChecked = snapshot.checked[itemId] != null
        const item = itemsById.get(itemId)

        marker.setIcon(categoryIcon(item.category, variant))
      })
    })

    // ── Cleanup ───────────────────────────────────────────────────────────
    return () => {
      unsubscribe()
      map.remove()

  }, []) // run once on mount

  // changes which items are in scope) — handled via an outer effect that
  // calls refreshItems() on each render, only if the map is already initialized.
  useEffect(() => {
    const map = mapRef.current

    const polyGroup = polyGroupRef.current
    if (!map || !itemGroup || !polyGroup) return

    markerMapRef.current.clear()

    polyGroup.clearLayers()
    renderPolylines(polyGroup, layerRef.current)
  }) // no dependency array — run after every render (but only does work if map initialized)

    const map = mapRef.current
    if (!map) return
    layerRef.current = code

    tileLayerRef.current?.setUrl(`${BASE_URL}tiles/${code}/{z}/{x}/{y}.webp`)

    gracesGroupRef.current?.clearLayers()
    renderGraces(gracesGroupRef.current!, code)

    itemGroupRef.current?.clearLayers()
    markerMapRef.current.clear()
    renderItems(itemGroupRef.current!, markerMapRef.current, code, map)

    polyGroupRef.current?.clearLayers()
    renderPolylines(polyGroupRef.current!, code)

    // Update tab active state
    tabContainerRef.current?.querySelectorAll<HTMLButtonElement>('.er-layer-tab[data-code]').forEach((btn) => {

  }

  return (
    <div
      ref={containerRef}
      className="h-full w-full bg-bg"
      style={{ background: '#0e0c09' }}
    />
  )

// ── Helpers (module-level, no React) ──────────────────────────────────────

function renderGraces(group: L.LayerGroup, code: MapCode) {
  for (const grace of graces) {
    if (grace.code !== code) continue
    L.marker([grace.lat, grace.lng], { icon: graceIcon(), zIndexOffset: -100 })
      .bindTooltip(grace.name, { direction: 'top', offset: [0, -10] })
      .addTo(group)
  }
}

function renderItems(
  group: L.LayerGroup,
  markerMap: Map<string, L.Marker>,
  code: MapCode,
  map: L.Map,
) {
  const snapshot = progressStore.getSnapshot()
  const nextUpId = findNextUpId()

  for (const region of regions) {
    // Collect all items for this region (legs + cleanup)
    const allSteps = [
      ...region.legs.flatMap((l) => l.steps),
      ...region.cleanup.map((id) => ({ type: 'item' as const, itemId: id })),
    ]
    for (const step of allSteps) {
      if (step.type !== 'item') continue
      const item = itemsById.get(step.itemId)
      if (!item?.map || item.map.code !== code) continue
      if (markerMap.has(step.itemId)) continue // dedup (item may appear in multiple legs)

      const isChecked = snapshot.checked[item.id] != null
      const isNextUp  = item.id === nextUpId
      const variant: MarkerVariant = isNextUp ? 'nextup' : isChecked ? 'checked' : 'normal'

      const marker = L.marker([item.map.lat, item.map.lng], {
        icon: categoryIcon(item.category, variant),
        zIndexOffset: isNextUp ? 200 : isChecked ? -50 : 0,
      })

      marker.bindPopup(() => buildPopup(item.id, map), {
        maxWidth: 290,
        minWidth: 280,
        autoPan: true,
      })

      marker.addTo(group)
      markerMap.set(item.id, marker)
    }
  }
}

function renderPolylines(group: L.LayerGroup, code: MapCode) {
  for (const region of regions) {
    for (const leg of region.legs) {
      const coords = routePathForLayer(leg, itemsById, code)
      if (coords.length < 2) continue
      L.polyline(coords, {
        color: '#c8a55a',
        weight: 2,
        opacity: 0.45,
        dashArray: undefined,
      }).addTo(group)
    }
  }
}

function findNextUpId(): string | null {
  const snapshot = progressStore.getSnapshot()
  for (const region of regions) {
    for (const leg of region.legs) {
      for (const step of leg.steps) {
        if (step.type !== 'item') continue
        if (snapshot.checked[step.itemId] == null) return step.itemId
      }
    }
  }
  return null
}
```

```typescript
      <span className="text-[10px] text-ink-dim whitespace-nowrap">{label}</span>
      <span className="h-1.5 flex-1 overflow-hidden rounded bg-edge">
        <span className="block h-full bg-gold transition-all" style={{ width: `${pct}%` }} />
      </span>
      <span className="text-[10px] text-ink-dim whitespace-nowrap">{done}/{total}</span>
    </div>
  )
}

// ── Step row inside RoutePanel ────────────────────────────────────────────

function PanelStepRow({
  step,
  isNextUp,
  onCheck,
  onLocate,
}: {
  step: Step
  isNextUp: boolean
  onCheck: (id: string) => void
  onLocate?: () => void
}) {
  const { snapshot } = useProgress()

  if (step.type === 'direction') {
    return <li className="py-1 pl-6 text-xs text-ink-dim italic">{step.text}</li>
  }

  const id = checkableId(step)
  const checked = id != null && snapshot.checked[id] != null
  const item = step.type === 'item' ? itemsById.get(step.itemId) : undefined
  const missable = step.type === 'quest' ? step.missable : item?.missable
  const name =
    step.type === 'item' ? (item?.name ?? step.itemId) :
    step.type === 'boss' ? step.text :
    step.text

  return (
    <li
      className={`flex items-start gap-2 rounded px-2 py-1.5 ${
        checked ? 'opacity-40' : isNextUp ? 'bg-panel2 border border-edge' : ''
      } hover:bg-panel2 transition-colors`}
    >
      {id != null && (
        <input
          type="checkbox"
          checked={checked}
          onChange={() => onCheck(id)}
          className="mt-0.5 size-3.5 accent-gold shrink-0"
          aria-label={name}
        />
      )}
      <div className="min-w-0 flex-1 text-xs">
        <span className={`${checked ? 'line-through text-ink-dim' : ''} ${isNextUp ? 'text-gold font-medium' : ''}`}>
          {step.type === 'boss' && '⚔ '}
          {step.type === 'quest' && `❖ ${step.questline}: `}
          {name}
        </span>
        {item && (
          <span className="ml-1.5 rounded border border-edge px-1 py-px text-[9px] text-ink-dim">
            {CATEGORY_META[item.category].label}
          </span>
        )}
        {item?.dlc && <span className="ml-1 text-[9px] text-gold-dim">DLC</span>}
        {(step.type === 'item' && (step.note ?? item?.acquisition)) && (
          <p className="mt-0.5 text-[10px] text-ink-dim leading-relaxed">
            {step.note ?? item?.acquisition}
          </p>
        )}
        {missable && (
          <p className="mt-0.5 text-[10px] font-semibold text-missable">
            ⚠ MISSABLE — {missable.lockedBy}
          </p>
        )}
      </div>
      {item?.map && onLocate && (
        <button
          onClick={onLocate}
          className="shrink-0 text-[10px] text-gold hover:underline"
          aria-label={`Locate ${name} on map`}
        >
          ⌖
        </button>
      )}
    </li>
  )
}

function RegionTree({

  activeRegionId: string

                return (
                  <NavLink

                    onClick={onClose}

                  </NavLink>
                )
              })}
          </div>
        )
      })}
    </div>
  )
}

// ── Main RoutePanel component ─────────────────────────────────────────────

export default function RoutePanel() {
  const { regionId, legId } = useParams()
  const navigate = useNavigate()
  const { snapshot, store } = useProgress()
  const { focus } = useUi()
  const [showTree, setShowTree] = useState(false)
  const [mobileExpanded, setMobileExpanded] = useState(false)

  const region = regions.find((r) => r.id === regionId)
  if (!region) {
    return (
      <div className="er-route-panel p-4 text-sm text-ink-dim">
        Select a region to begin.
      </div>
    )
  }

  // ── Leg navigation ────────────────────────────────────────────────────
  const currentLeg = legId ? region.legs.find((l) => l.id === legId) ?? null : null
  const currentLegIndex = currentLeg ? region.legs.indexOf(currentLeg) : -1
  const prevLeg = currentLegIndex > 0 ? region.legs[currentLegIndex - 1] : null
  const nextLeg = currentLegIndex < region.legs.length - 1 ? region.legs[currentLegIndex + 1] : null

  // ── Steps to show ─────────────────────────────────────────────────────
  const displaySteps: Step[] = currentLeg
    ? currentLeg.steps
    : [
        ...region.legs.flatMap((l) => l.steps),
        ...region.cleanup.map((id): Step => ({ type: 'item', itemId: id })),
      ]

  // ── Find next up ──────────────────────────────────────────────────────
  const nextUpStep = displaySteps.find((step) => {
    const id = checkableId(step)
    if (id == null) return false
    return snapshot.checked[id] == null
  }) ?? null

  // ── Handle check from "next up" button ───────────────────────────────
  function handleCheck(id: string) {
    store.toggle(id)
    // After checking, find the new next-up and pan to it
    setTimeout(() => {
      const newSnapshot = store.getSnapshot()
      for (const step of displaySteps) {
        const sid = checkableId(step)
        if (sid == null) continue
        if (newSnapshot.checked[sid] != null) continue
        // First unchecked after toggle — pan to it if it has a map ref
        if (step.type === 'item') {
          const item = itemsById.get(step.itemId)
          if (item?.map) focus(item.map, item.id)
        }
        break
      }
    }, 50)
  }

  // ── Region progress ───────────────────────────────────────────────────
  const regionIds = regionCheckables(region)
  const regionDone = countChecked(regionIds, snapshot.checked)
  const legIds = currentLeg ? legCheckables(currentLeg) : []
  const legDone = currentLeg ? countChecked(legIds, snapshot.checked) : 0

  return (
    <div className={`er-route-panel relative ${mobileExpanded ? 'er-route-panel--expanded' : ''}`}>
      {/* Drag handle (mobile) */}
      <div
        className="er-drag-handle sm:hidden"
        onClick={() => setMobileExpanded(!mobileExpanded)}
        title="Toggle panel"
      />

      {/* Header */}
      <div className="sticky top-0 z-[1] border-b border-edge bg-panel px-3 py-2 shrink-0">
        <div className="flex items-center gap-2">
          <button
            onClick={() => setShowTree(true)}
            className="text-[10px] text-ink-dim hover:text-gold rounded border border-edge px-1.5 py-0.5"
            title="All regions"
          >
            ≡
          </button>
          <h1 className="font-display text-sm text-gold flex-1 truncate">{region.name}</h1>
          {region.dlc && <span className="text-[9px] text-gold-dim border border-gold-dim/40 rounded px-1">DLC</span>}
        </div>
        {/* Leg navigation */}
        {region.legs.length > 0 && (
          <div className="mt-1.5 flex items-center gap-1 text-[10px]">
            <button
              disabled={!prevLeg}
              onClick={() => prevLeg && navigate(`/region/${region.id}/${prevLeg.id}`)}
              className="rounded border border-edge px-1.5 py-0.5 text-ink-dim hover:text-gold disabled:opacity-30 disabled:cursor-not-allowed"
            >
              ‹
            </button>
            <span className="flex-1 truncate text-center text-ink-dim">
              {currentLeg
                ? `${currentLeg.from} → ${currentLeg.to}`
                : 'Region sweep'}
            </span>
            <button
              disabled={!nextLeg}
              onClick={() => nextLeg && navigate(`/region/${region.id}/${nextLeg.id}`)}
              className="rounded border border-edge px-1.5 py-0.5 text-ink-dim hover:text-gold disabled:opacity-30 disabled:cursor-not-allowed"
            >
              ›
            </button>
          </div>
        )}
        {/* Progress bars */}
        <div className="mt-1.5 space-y-0.5">
          {currentLeg && <ProgressBar done={legDone} total={legIds.length} label="Leg" />}
          <ProgressBar done={regionDone} total={regionIds.length} label="Region" />
        </div>
      </div>

      {/* Next up callout */}
      {nextUpStep && (
        <NextUpCallout
          step={nextUpStep}
          leg={currentLeg}
          onCheck={handleCheck}
          onLocate={() => {
            if (nextUpStep.type === 'item') {
              const item = itemsById.get(nextUpStep.itemId)
              if (item?.map) focus(item.map, item.id)
            }
          }}
        />
      )}

      {/* Running list */}
      <div className="flex-1 overflow-y-auto px-1 pb-4">
        {currentLeg?.summary && (
          <p className="px-3 py-2 text-[11px] text-ink-dim italic leading-relaxed border-b border-edge/50">
            {currentLeg.summary}
          </p>
        )}
        <ul className="space-y-0.5 py-1">
          {displaySteps.map((step, i) => {
            const id = checkableId(step)
            const isNextUp = nextUpStep != null && id != null && checkableId(nextUpStep) === id
            const item = step.type === 'item' ? itemsById.get(step.itemId) : undefined
            return (
              <PanelStepRow
                key={i}
                step={step}
                isNextUp={isNextUp}
                onCheck={handleCheck}
                onLocate={
                  item?.map ? () => focus(item.map!, item.id) : undefined
                }
              />
            )
          })}
        </ul>
        {!currentLeg && region.cleanup.length === 0 && region.legs.length === 0 && (
          <p className="px-3 py-4 text-xs text-ink-dim">No steps authored yet for this region.</p>
        )}
      </div>

      {/* Region tree drawer */}
      {showTree && (
        <RegionTree activeRegionId={region.id} onClose={() => setShowTree(false)} />
      )}
    </div>
  )
}
```
