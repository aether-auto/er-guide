```typescript
// ── Config ────────────────────────────────────────────────────────────────────

const SYNC_CONFIG_KEY = 'er-guide-gist-sync'
const GIST_FILENAME = 'er-guide-progress.json'
const GIST_DESCRIPTION = 'Elden Ring 100% guide progress — synced by er-guide'

export interface GistSyncConfig {
  token: string
  gistId: string
  lastSyncedAt: string | null
}

export type SyncStatus =
  | { type: 'disconnected' }
  | { type: 'connecting' }
  | { type: 'idle'; gistId: string; lastSyncedAt: string | null }
  | { type: 'syncing' }
  | { type: 'cloud-newer'; gistId: string; lastSyncedAt: string | null }
  | { type: 'error'; message: string; gistId?: string; lastSyncedAt?: string | null }

// ── Pure helpers ──────────────────────────────────────────────────────────────

export function parseConfig(raw: string | null): GistSyncConfig | null {
  if (!raw) return null
  try {
    const parsed = JSON.parse(raw)
    if (
      parsed == null ||
      typeof parsed !== 'object' ||
      typeof parsed.token !== 'string' ||
      typeof parsed.gistId !== 'string'
    ) return null
    if (!('lastSyncedAt' in parsed)) return null
    return { token: parsed.token, gistId: parsed.gistId, lastSyncedAt: parsed.lastSyncedAt ?? null }
  } catch {
    return null
  }
}

export function serializeConfig(cfg: GistSyncConfig): string {
  return JSON.stringify(cfg)
}

export function isGistNewer(gistUpdatedAt: string, lastSyncedAt: string | null): boolean {
  if (lastSyncedAt === null) return false
  return new Date(gistUpdatedAt).getTime() > new Date(lastSyncedAt).getTime()
}
```

```typescript
export interface GistFileContent {
  content: string
  updatedAt: string
}

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export function extractGistContent(gistResponse: any): GistFileContent | null {
  const file = gistResponse?.files?.[GIST_FILENAME]
  if (!file?.content) return null
  return { content: file.content, updatedAt: gistResponse.updated_at as string }
}

export function buildGistPayload(content: string, isUpdate: boolean): object {
  if (isUpdate) {
    return { files: { [GIST_FILENAME]: { content } } }
  }
  return {
    description: GIST_DESCRIPTION,
    public: false,
    files: { [GIST_FILENAME]: { content } },
  }
}
```

```typescript
// ── Debounce with injectable timer ────────────────────────────────────────────

type SetTimeoutFn = (fn: () => void, ms: number) => ReturnType<typeof setTimeout>
type ClearTimeoutFn = (id: ReturnType<typeof setTimeout> | undefined) => void

export function createDebouncer(
  fn: () => void,
  delayMs: number,
  setTimeoutFn: SetTimeoutFn = setTimeout,
  clearTimeoutFn: ClearTimeoutFn = clearTimeout,
) {
  let timer: ReturnType<typeof setTimeout> | undefined

  function debounced() {
    clearTimeoutFn(timer)
    timer = setTimeoutFn(() => {
      timer = undefined
      fn()
    }, delayMs)
  }

  debounced.cancel = () => {
    clearTimeoutFn(timer)
    timer = undefined
  }

  return debounced
}
```

```typescript
// ── createGistSync factory ────────────────────────────────────────────────────

type StorageLike = Pick<Storage, 'getItem' | 'setItem' | 'removeItem'>

export function createGistSync(storage: StorageLike = localStorage) {
  let config: GistSyncConfig | null = parseConfig(storage.getItem(SYNC_CONFIG_KEY))
  let status: SyncStatus = config
    ? { type: 'idle', gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
    : { type: 'disconnected' }
  const listeners = new Set<() => void>()

  function notify() {
    listeners.forEach((fn) => fn())
  }

  function setStatus(s: SyncStatus) {
    status = s
    notify()
  }

  function saveConfig(cfg: GistSyncConfig) {
    config = cfg
    storage.setItem(SYNC_CONFIG_KEY, serializeConfig(cfg))
  }

  async function githubFetch(
    path: string,
    token: string,
    options: RequestInit = {},
  ): Promise<{ ok: boolean; status: number; data: unknown }> {
    const res = await fetch(`https://api.github.com${path}`, {
      ...options,
      headers: {
        Authorization: `Bearer ${token}`,
        Accept: 'application/vnd.github+json',
        'X-GitHub-Api-Version': '2022-11-28',
        'Content-Type': 'application/json',
        ...(options.headers ?? {}),
      },
    })
    const data = res.ok ? await res.json() : null
    return { ok: res.ok, status: res.status, data }
  }

  async function findExistingGist(token: string): Promise<string | null> {
    const result = await githubFetch('/gists', token)
    if (!result.ok) return null
    const list = result.data as Array<{ id: string; description: string; files: Record<string, unknown> }>
    const match = list.find(
      (g) =>
        g.description === GIST_DESCRIPTION &&
        g.files[GIST_FILENAME] != null,
    )
    return match?.id ?? null
  }

  return {
    subscribe(fn: () => void) {
      listeners.add(fn)
      return () => void listeners.delete(fn)
    },

    getStatus(): SyncStatus {
      return status
    },

    async connect(token: string): Promise<SyncStatus> {
      setStatus({ type: 'connecting' })
      try {
        // Validate token: list gists
        const listResult = await githubFetch('/gists', token)
        if (!listResult.ok) {
          const s: SyncStatus = { type: 'error', message: `GitHub returned ${listResult.status} — check your token` }
          setStatus(s)
          return s
        }

        // Find or create gist
        const list = listResult.data as Array<{ id: string; description: string; files: Record<string, unknown> }>
        let gistId = list.find(
          (g) => g.description === GIST_DESCRIPTION && g.files[GIST_FILENAME] != null,
        )?.id ?? null

        if (!gistId) {
          // Create new secret gist with empty content
          const createResult = await githubFetch('/gists', token, {
            method: 'POST',
            body: JSON.stringify(buildGistPayload('{}', false)),
          })
          if (!createResult.ok) {
            const s: SyncStatus = { type: 'error', message: 'Failed to create gist' }
            setStatus(s)
            return s
          }
          gistId = (createResult.data as { id: string }).id
        }

        saveConfig({ token, gistId, lastSyncedAt: null })
        const s: SyncStatus = { type: 'idle', gistId, lastSyncedAt: null }
        setStatus(s)
        return s
      } catch (err) {
        const message = err instanceof Error ? err.message : 'Unknown error'
        const s: SyncStatus = { type: 'error', message }
        setStatus(s)
        return s
      }
    },

    disconnect() {
      config = null
      storage.removeItem(SYNC_CONFIG_KEY)
      setStatus({ type: 'disconnected' })
    },
  }
}
```

```typescript
    async pushNow(saveJson: string): Promise<SyncStatus> {
      if (!config) {
        const s: SyncStatus = { type: 'error', message: 'Not connected' }
        setStatus(s)
        return s
      }
      setStatus({ type: 'syncing' })
      try {
        const result = await githubFetch(`/gists/${config.gistId}`, config.token, {
          method: 'PATCH',
          body: JSON.stringify(buildGistPayload(saveJson, true)),
        })
        if (!result.ok) {
          const s: SyncStatus = { type: 'error', message: `Push failed (${result.status})`, gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
          setStatus(s)
          return s
        }
        const updatedAt = (result.data as { updated_at: string }).updated_at
        saveConfig({ ...config, lastSyncedAt: updatedAt })
        const s: SyncStatus = { type: 'idle', gistId: config.gistId, lastSyncedAt: updatedAt }
        setStatus(s)
        return s
      } catch (err) {
        const message = err instanceof Error ? err.message : 'Unknown error'
        const s: SyncStatus = { type: 'error', message, gistId: config?.gistId, lastSyncedAt: config?.lastSyncedAt }
        setStatus(s)
        return s
      }
    },

    async pullNow(importFn: (json: string) => void): Promise<SyncStatus> {
      if (!config) {
        const s: SyncStatus = { type: 'error', message: 'Not connected' }
        setStatus(s)
        return s
      }
      setStatus({ type: 'syncing' })
      try {
        const result = await githubFetch(`/gists/${config.gistId}`, config.token)
        if (!result.ok) {
          const s: SyncStatus = { type: 'error', message: `Pull failed (${result.status})`, gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
          setStatus(s)
          return s
        }
        const file = extractGistContent(result.data)
        if (file) {
          importFn(file.content)
          saveConfig({ ...config, lastSyncedAt: file.updatedAt })
        }
        const s: SyncStatus = { type: 'idle', gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
        setStatus(s)
        return s
      } catch (err) {
        const message = err instanceof Error ? err.message : 'Unknown error'
        const s: SyncStatus = { type: 'error', message, gistId: config?.gistId, lastSyncedAt: config?.lastSyncedAt }
        setStatus(s)
        return s
      }
    },

    async checkCloudNewer(): Promise<SyncStatus> {
      if (!config) return status
      try {
        const result = await githubFetch(`/gists/${config.gistId}`, config.token)
        if (!result.ok) return status
        const file = extractGistContent(result.data)
        if (!file) return status
        if (isGistNewer(file.updatedAt, config.lastSyncedAt)) {
          const s: SyncStatus = { type: 'cloud-newer', gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
          setStatus(s)
          return s
        }
        const s: SyncStatus = { type: 'idle', gistId: config.gistId, lastSyncedAt: config.lastSyncedAt }
        setStatus(s)
        return s
      } catch {
        // offline — silently stay local
        return status
      }
    },
```

```markdown
## Cloud sync (optional)

Progress data can be backed up to a private GitHub Gist and synced across browsers. No server required — the site talks directly to the GitHub Gist API from your browser.

| Data | Location |

| Your GitHub token + gist ID | Browser `localStorage` under `er-guide-gist-sync` — never leaves your device except in requests to `api.github.com` |
| Cloud backup | A **secret** GitHub Gist in your account — only accessible with your token |

Disconnecting removes the token from your browser but **does not delete the gist**. You can reconnect later and it will be found automatically.
```
